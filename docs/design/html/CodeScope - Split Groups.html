> implement CSV export for reports
Writing CsvWriter helper and wiring the endpoint on ReportsController.
// Stream a CSV of the report so we never page into memory.
[HttpGet("{id}/export.csv")]
public async Task<IActionResult> ExportCsv(Guid id, CancellationToken ct)
{
var report = await _reports.GetAsync(id, ct);
if (report is null) return NotFound();
var sb = new StringBuilder();
sb.AppendLine("Site,Technician,Date,Finding,Severity");
foreach (var row in report.Findings)
{
sb.Append(Csv.Escape(row.Site)).Append(",");
sb.Append(row.Date.ToString("yyyy-MM-dd")).Append(",");
sb.AppendLine(row.Severity.ToString());
}
return File(Encoding.UTF8.GetBytes(sb.ToString()), "text/csv");
}
